\newcommand{\qnote}[3]{%
  \pgfmathsetmacro{\x}{2.35 + (#1 - 1) * 0.76}
  \pgfmathsetmacro{\y}{\pitchy{#2}}
  \fill[rotate around={-18:(\x,\y)}] (\x,\y) ellipse (0.15 and 0.11);
  \ifdim \y pt > 1.75pt
    \draw[line width=0.8pt] (\x - 0.12,\y) -- (\x - 0.12,\y - 1.35);
  \else
    \draw[line width=0.8pt] (\x + 0.12,\y) -- (\x + 0.12,\y + 1.35);
  \fi
  \node[font=\tiny, align=center] at (\x,-0.78) {#3};
}
