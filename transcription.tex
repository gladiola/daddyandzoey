\newcommand{\drawstaffbase}{%
  \foreach \y in {0,0.5,1,1.5,2} {
    \draw[line width=0.4pt] (0,\y) -- (\staffright,\y);
  }
  \drawbassclef
  \node[font=\small] at (1.55,1.00) {$\mathbf{\frac{4}{4}}$};
  \pgfmathtruncatemacro{\internalbars}{\measuresperline - 1}
  \foreach \measure in {1,...,\internalbars} {
    \pgfmathsetmacro{\barx}{\staffleft + (\measure * \beatspermeasure - 0.5) * \beatspacing}
    \draw[line width=0.6pt] (\barx,-0.12) -- (\barx,2.12);
  }
  \pgfmathsetmacro{\finalbarx}{\staffleft + (\beatsperline - 0.5) * \beatspacing}
  \draw[line width=1.0pt] (\finalbarx,-0.12) -- (\finalbarx,2.12);
}

\newcommand{\drawaccidental}[3]{%
  \def\accidentaltype{#3}%
  \ifx\accidentaltype\@empty\else
    \def\sharpsymbol{sharp}%
    \def\flatsymbol{flat}%
    \def\naturalsymbol{natural}%
    \ifx\accidentaltype\sharpsymbol
      \node[font=\scriptsize] at (#1 - 0.30,#2) {$\sharp$};
    \else\ifx\accidentaltype\flatsymbol
      \node[font=\scriptsize] at (#1 - 0.28,#2) {$\flat$};
    \else\ifx\accidentaltype\naturalsymbol
      \node[font=\scriptsize] at (#1 - 0.28,#2) {$\natural$};
    \fi\fi\fi
  \fi
}

\newcommand{\drawledgerlines}[2]{%
  \pgfmathtruncatemacro{\belowcount}{max(0, int(floor((0 - (#2) - 0.5) / 0.5)) + 1)}
  \ifnum\belowcount>0
    \foreach \step in {1,...,\belowcount} {
      \pgfmathsetmacro{\ledgery}{0 - 0.5 * \step}
      \draw[line width=0.5pt] (#1 - 0.26,\ledgery) -- (#1 + 0.26,\ledgery);
    }
  \fi
  \pgfmathtruncatemacro{\abovecount}{max(0, int(floor(((#2) - 2 - 0.5) / 0.5)) + 1)}
  \ifnum\abovecount>0
    \foreach \step in {1,...,\abovecount} {
      \pgfmathsetmacro{\ledgery}{2 + 0.5 * \step}
      \draw[line width=0.5pt] (#1 - 0.26,\ledgery) -- (#1 + 0.26,\ledgery);
    }
  \fi
}

\newcommand{\drawstemandflags}[3]{%
  \pgfmathparse{#2 > 1 ? 1 : 0}%
  \ifnum\pgfmathresult=1
    \draw[line width=0.8pt] (#1 - 0.12,#2) -- (#1 - 0.12,#2 - 1.35);
    \ifdim #3pt < 1pt
      \draw[line width=0.8pt]
        (#1 - 0.12,#2 - 1.35)
        .. controls (#1 + 0.18,#2 - 1.12) and (#1 + 0.10,#2 - 0.88) .. (#1 - 0.02,#2 - 0.76);
    \fi
  \else
    \draw[line width=0.8pt] (#1 + 0.12,#2) -- (#1 + 0.12,#2 + 1.35);
    \ifdim #3pt < 1pt
      \draw[line width=0.8pt]
        (#1 + 0.12,#2 + 1.35)
        .. controls (#1 + 0.42,#2 + 1.12) and (#1 + 0.34,#2 + 0.88) .. (#1 + 0.22,#2 + 0.76);
    \fi
  \fi
}

\newcommand{\drawnotesymbol}[4]{%
  \pgfmathsetmacro{\x}{\staffleft + (#1) * \beatspacing}
  \pgfmathsetmacro{\y}{\pitchy{#2}}
  \def\noteduration{#3}%
  \drawledgerlines{\x}{\y}
  \drawaccidental{\x}{\y}{\pitchaccidental{#2}}
  \ifdim \noteduration pt > 3pt
    \draw[rotate around={-18:(\x,\y)}, line width=0.8pt] (\x,\y) ellipse (0.18 and 0.12);
  \else\ifdim \noteduration pt > 1pt
    \draw[rotate around={-18:(\x,\y)}, line width=0.8pt] (\x,\y) ellipse (0.16 and 0.11);
    \drawstemandflags{\x}{\y}{\noteduration}
  \else
    \fill[rotate around={-18:(\x,\y)}] (\x,\y) ellipse (0.15 and 0.11);
    \drawstemandflags{\x}{\y}{\noteduration}
  \fi\fi
  \node[font=\tiny, align=center] at (\x,-1.00) {#4};
}

\newcommand{\musicline}[1]{%
  \begin{center}
    \begin{tikzpicture}[x=1cm,y=1cm]
      \drawstaffbase

    \end{tikzpicture}
  \end{center}
}

\newcommand{\resetmusicstate}{%
  \gdef\currentmusiccommands{}%
  \xdef\linebeat{0}%
}

\newcommand{\flushmusicline}{%
  \ifx\currentmusiccommands\@empty\else
    \musicline{\currentmusiccommands}%
    \gdef\currentmusiccommands{}%
    \xdef\linebeat{0}%
  \fi
}

\newcommand{\ensurelinecapacity}[1]{%
  \pgfmathparse{\linebeat + (#1) > \beatsperline ? 1 : 0}%
  \ifnum\pgfmathresult=1
    \flushmusicline
  \fi
}

\newcommand{\validateeventduration}[1]{%
  \pgfmathparse{#1 > \beatsperline ? 1 : 0}%
  \ifnum\pgfmathresult=1
    \PackageError{transcription}{Event duration #1 exceeds beats per line \beatsperline}{Reduce the event duration or increase \string\beatsperline.}%
  \fi
}

\newcommand{\advancebeat}[1]{%
  \pgfmathparse{\linebeat + (#1)}%
  \xdef\linebeat{\pgfmathresult}%
}

\newcommand{\NoteEvent}[3]{%
  \validateeventduration{#2}%
  \ensurelinecapacity{#2}%
  \edef\eventstart{\linebeat}%
  \begingroup
  \edef\next{%
    \endgroup
    \noexpand\g@addto@macro\noexpand\currentmusiccommands{%
      \noexpand\drawnotesymbol{\eventstart}{#1}{#2}{#3}%
    }%
  }%
  \next
  \advancebeat{#2}%
}

\newcommand{\RestEvent}[2]{%
  \validateeventduration{#1}%
  \ensurelinecapacity{#1}%
  \edef\eventstart{\linebeat}%
  \begingroup
  \edef\next{%
    \endgroup
    \noexpand\g@addto@macro\noexpand\currentmusiccommands{%
      \noexpand\drawrestsymbol{\eventstart}{#1}{#2}%
    }%
  }%
  \next
  \advancebeat{#1}%
}

\newcommand{\RenderTranscription}[1]{%
  \resetmusicstate
  #1%
  \flushmusicline
}
